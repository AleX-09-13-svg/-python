--------------------------------------------------------+------------------------------------------------------------
import pythoncom                                        | Подключает модуль COM для связи Python с Windows-программами.
from win32com.client import CastTo, dynamic              | Берет инструменты pywin32 для работы с объектами Inventor.
                                                        |
HOLE_DIAMETER = "5 mm"                                  | Диаметр будущего отверстия.
HOLE_DEPTH = "10 mm"                                    | Глубина будущего отверстия.
                                                        |
POSITIVE_EXTENT_DIRECTION = 20993                       | Константа Inventor: направление отверстия по нормали эскиза.
                                                        |
def get_inventor():                                     | Функция подключения к уже открытому Autodesk Inventor.
    return dynamic.Dispatch(                            | Оборачивает найденное приложение Inventor в COM-объект Python.
        pythoncom.GetActiveObject("Inventor.Application").QueryInterface(
                                                        | Находит уже запущенный Inventor и получает его интерфейс.
            pythoncom.IID_IDispatch                     | Указывает стандартный COM-интерфейс для управления объектом.
        )                                               |
    )                                                   |
                                                        |
def face_center_in_sketch(sketch, face, tg):            | Функция вычисляет центр выбранной грани в координатах эскиза.
    points = [                                          | Создает список точек-вершин грани.
        sketch.ModelToSketchSpace(face.Vertices.Item(index).Point)
                                                        | Переводит каждую 3D-вершину грани в 2D-координаты эскиза.
        for index in range(1, face.Vertices.Count + 1)  | Перебирает вершины Inventor; нумерация начинается с 1.
    ]                                                   |
                                                        |
    return tg.CreatePoint2d(                            | Создает 2D-точку центра.
        sum(point.X for point in points) / len(points), | Среднее значение X всех вершин.
        sum(point.Y for point in points) / len(points), | Среднее значение Y всех вершин.
    )                                                   |
                                                        |
def object_collection(inv, objects):                    | Функция создает коллекцию объектов Inventor.
    collection = inv.TransientObjects.CreateObjectCollection()
                                                        | Создает пустую временную коллекцию.
    for obj in objects:                                 | Перебирает переданные объекты.
        collection.Add(obj)                             | Добавляет каждый объект в коллекцию.
    return collection                                   | Возвращает готовую коллекцию.
                                                        |
def main():                                             | Главная функция скрипта.
    pythoncom.CoInitialize()                            | Инициализирует COM перед работой с Inventor.
                                                        |
    inv = get_inventor()                                | Получает объект запущенного Inventor.
    doc = CastTo(inv.ActiveDocument, "PartDocument")    | Берет активный документ и приводит его к типу PartDocument.
    comp = doc.ComponentDefinition                      | Получает содержимое детали: тела, эскизы, операции.
                                                        |
    face = doc.SelectSet.Item(1)                        | Берет первый выбранный объект; здесь ожидается выбранная грань.
                                                        |
    sketch = comp.Sketches.Add(face)                    | Создает новый 2D-эскиз на выбранной грани.
    center = face_center_in_sketch(sketch, face, inv.TransientGeometry)
                                                        | Находит центр выбранной грани в координатах эскиза.
    sketch_point = sketch.SketchPoints.Add(center, False)
                                                        | Добавляет точку в центр грани.
    sketch_point.HoleCenter = True                      | Помечает точку как центр отверстия.
    sketch.Solve()                                      | Пересчитывает эскиз.
                                                        |
    hole_features = comp.Features.HoleFeatures          | Получает коллекцию операций "Отверстие".
    placement = hole_features.CreateSketchPlacementDefinition(
                                                        | Создает описание размещения отверстия по точке эскиза.
        object_collection(inv, [sketch_point])          | Передает точку центра как коллекцию объектов.
    )                                                   |
    hole_features.AddDrilledByDistanceExtent(           | Создает сверленое отверстие заданной глубины.
        placement,                                      | Где расположить отверстие.
        HOLE_DIAMETER,                                  | Диаметр отверстия.
        HOLE_DEPTH,                                     | Глубина отверстия.
        POSITIVE_EXTENT_DIRECTION,                      | Направление отверстия.
        True,                                           | Плоское дно отверстия.
    )                                                   |
                                                        |
    doc.Update()                                        | Обновляет модель детали после создания операции.
                                                        |
if __name__ == "__main__":                              | Запускает main(), если файл запущен как скрипт.
    main()                                              | Старт программы.
